\section{The Translation}\label{sec:translation}

\AP Fix a formula $\alpha$. We assign a \intro{polarity}, \intro{positive}
or \intro{negative}, to each occurrence in $\alpha$ as follows. The root
$\rt$ is \kl{positive}. If
$\subf{\alpha}{u}=\subf{\alpha}{u0}\to\subf{\alpha}{u1}$, then $u1$ has
the same \kl{polarity} as $u$ and $u0$ the opposite one. \AP For each
\kl{non-leaf occurrence} $u$ we choose a new variable $\intro*\ev{u}$,
called an \intro{extension variable}, so that distinct occurrences receive
distinct variables and no $\ev{u}$ occurs in $\alpha$. For every
occurrence $u$ let
\[
\intro*\rv{u}=\begin{cases}\subf{\alpha}{u},&\text{if $u$ is a \kl{leaf}},\\
\ev{u},&\text{if $u$ is a \kl{non-leaf occurrence}.}\end{cases}
\]
\AP For every \kl{non-leaf occurrence} $u$ we put
$\intro*\body{u}=\rv{u0}\to\rv{u1}$ and
\[
\intro*\dprem{u}=\begin{cases}\body{u}\to\ev{u},&\text{if $u$ is \kl{positive}},\\
\ev{u}\to\body{u},&\text{if $u$ is \kl{negative}.}\end{cases}
\]

\begin{definition}\label{def:S}
\AP Let $u_1,\dots,u_m$ be the \kl{non-leaf occurrences} of $\alpha$,
listed in post-order; that is, $u_i$ precedes $u_j$ if and only if $u_i$
is \kl{strictly below} $u_j$, or $u_i=w0u'$ and $u_j=w1u''$ for some
words $w,u',u''$. The \intro{translation} of $\alpha$ is the formula
\[
\intro*\St(\alpha)=\dprem{u_1}\to\dots\to\dprem{u_m}\to\rv{\rt} .
\]
\end{definition}

If $\alpha$ is a variable, then $m=0$ and $\St(\alpha)=\alpha$.

\begin{lemma}\label{lem:order}
$\Ord(\St(\alpha))\le4$ for every formula $\alpha$.
\end{lemma}
\begin{proof}
Every formula $\body{u}=\rv{u0}\to\rv{u1}$ has \kl{order} $2$, so every
formula $\dprem{u}$ has \kl{order} at most $3$, and hence
$\Ord(\St(\alpha))\le4$.
\end{proof}

\AP Let $\intro*\decsub{\alpha}$ be the substitution that maps each
$\ev{u}$ to $\subf{\alpha}{u}$ and fixes all other variables; then
$\decsub{\alpha}(\rv{v})=\subf{\alpha}{v}$ for every occurrence $v$ of
$\alpha$.


\AP In examples we use the following notation, similar to that of
Jeřábek~\cite[Def.~3.6]{Jer16}. If $u$ is the only occurrence of a compound
formula $\gamma$ in $\alpha$ that is \kl{positive}, then we write
$\intro*\evfp{\gamma}$ for $\ev{u}$; if $u$ is the only occurrence of
$\gamma$ in $\alpha$ that is \kl{negative}, then we write
$\intro*\evfn{\gamma}$ for $\ev{u}$. In particular, $\evfp{\alpha}=\ev{\rt}$.

\begin{example}\label{ex:peirce}
For Peirce's law $\alpha=((a\to b)\to a)\to a$,
\[
\St(\alpha)=((a\to b)\to\evfp{a\to b})\to(\evfn{(a\to b)\to a}\to\evfp{a\to b}\to a)
\to((\evfn{(a\to b)\to a}\to a)\to\evfp{\alpha})\to\evfp{\alpha}.
\]
\end{example}

\begin{example}\label{ex:lc}
For the axiom $\alpha=((a\to b)\to c)\to((b\to a)\to c)\to c$ of
Gödel--Dummett logic,
\[
\begin{aligned}
\St(\alpha)={}&((a\to b)\to\evfp{a\to b})\to(\evfn{(a\to b)\to c}\to\evfp{a\to b}\to c)\\
&\to((b\to a)\to\evfp{b\to a})\to(\evfn{(b\to a)\to c}\to\evfp{b\to a}\to c)\\
&\to((\evfn{(b\to a)\to c}\to c)\to\evfp{((b\to a)\to c)\to c})\\
&\to((\evfn{(a\to b)\to c}\to\evfp{((b\to a)\to c)\to c})\to\evfp{\alpha})\to\evfp{\alpha}.
\end{aligned}
\]
\end{example}
