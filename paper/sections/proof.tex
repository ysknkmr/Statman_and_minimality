\section{Proof of the Main Theorem}\label{sec:proof}

Throughout this section, $\alpha$ is a formula that is not a variable,
$u_1,\dots,u_m$ are its \kl{non-leaf occurrences} in post-order, and $z$
is a variable not occurring in $\St(\alpha)$. By
Proposition~\ref{cor:criterion}, it suffices to show, for a \kl{logic}
$L$ containing $\axI$, that if $\Gen{\St(\alpha)}\cap L\neq\emptyset$, then
$\Gen{\alpha}\cap L\neq\emptyset$.

\AP We first name some occurrences in $\St(\alpha)$. For $1\le i\le m+1$,
let $\intro*\tl{i}=\dprem{u_i}\to\dots\to\dprem{u_m}\to\ev{\rt}$; thus
$\tl{1}=\St(\alpha)$, $\tl{m+1}=\ev{\rt}$, and $\tl{i}$ occurs in
$\St(\alpha)$ at $1^{i-1}$. For $u=u_i$, the formula $\dprem{u}$ occurs at
$\intro*\pprem(u)=1^{i-1}0$, and we let $\intro*\pbody(u)$ and
$\intro*\pdef(u)$ be the occurrences of $\body{u}$ and $\ev{u}$ in
$\dprem{u}$; that is, $\pbody(u)=\pprem(u)0$ and $\pdef(u)=\pprem(u)1$ if
$u$ is \kl{positive}, and $\pbody(u)=\pprem(u)1$ and $\pdef(u)=\pprem(u)0$
if $u$ is \kl{negative}. Finally, for an occurrence $v$ of $\alpha$ we
define the occurrence $\intro*\prep(v)$ of $\rv{v}$ in $\St(\alpha)$ by
$\prep(w0)=\pbody(w)0$, $\prep(w1)=\pbody(w)1$ and $\prep(\rt)=1^m$.

\begin{lemma}\label{lem:shape}
\leavevmode
\begin{enumerate}
\item[(a)] Every occurrence in $\St(\alpha)$ is exactly one of $1^{i-1}$
  $(1\le i\le m)$, $\pprem(u)$, $\pbody(u)$, $\pdef(u)$ and $\prep(v)$,
  and the subformula of $\St(\alpha)$ there is $\tl{i}$, $\dprem{u}$,
  $\body{u}$, $\ev{u}$ and $\rv{v}$, respectively.
\item[(b)] For a \kl{non-leaf occurrence} $u$ of $\alpha$, the variable
  $\ev{u}$ occurs in $\St(\alpha)$ exactly at $\pdef(u)$ and $\prep(u)$. A
  variable $c$ of $\alpha$ occurs in $\St(\alpha)$ exactly at the
  occurrences $\prep(v)$ with $v$ a \kl{leaf} of $\alpha$ and
  $\subf{\alpha}{v}=c$.
\item[(c)] $\hgt(\body{u})=1$, $\hgt(\dprem{u})=2$ and $\hgt(\tl{i})\ge3$
  for $i\le m$. Hence a formula of \kl{height}~$1$ occurs in $\St(\alpha)$
  only at occurrences $\pbody(u)$, a formula of \kl{height}~$2$ only at
  occurrences $\pprem(u)$, and a formula of \kl{height} at least~$3$ only
  at occurrences $1^{i-1}$.
\item[(d)] If $\body{u}=\body{v}$, then $\subf{\alpha}{u}=\subf{\alpha}{v}$.
  If $\dprem{u}=\dprem{v}$, then $u=v$. If $\tl{i}=\tl{j}$, then $i=j$.
\end{enumerate}
\end{lemma}
\begin{proof}
(a) and (b) follow from Definition~\ref{def:S}, since the
\kl{extension variables} are distinct and do not occur in $\alpha$.

(c) We have $\body{u}=\rv{u0}\to\rv{u1}$ with $\rv{u0}$ and $\rv{u1}$
variables, $\dprem{u}$ is $\body{u}\to\ev{u}$ or $\ev{u}\to\body{u}$, and
$\tl{i}=\dprem{u_i}\to\tl{i+1}$. The second claim follows from~(a), since
the formulas at $\pdef(u)$ and $\prep(v)$ are variables.

(d) If $\body{u}=\body{v}$, then
$\subf{\alpha}{u}=\decsub{\alpha}(\body{u})=\decsub{\alpha}(\body{v})=\subf{\alpha}{v}$.
Since $\body{u}$ is not a variable, $\dprem{u}$ is of the form
$\gamma\to p$ with $p$ a variable if and only if $u$ is \kl{positive}.
Hence $\dprem{u}=\dprem{v}$ implies that $u$ and $v$ have the same
\kl{polarity}, so $\ev{u}=\ev{v}$ and $u=v$. Finally, if $i<j$, then
$\tl{j}$ is a proper subformula of $\tl{i}$, so $\tl{i}\ne\tl{j}$.
\end{proof}

\AP Let $\intro*\Genb{\alpha}$ be the set of formulas
$\gen{\St(\alpha)}{P}{z}\in\Gen{\St(\alpha)}$ such that every element of $P$
is of the form $\pbody(u)$ or $\prep(v)$.

\begin{lemma}\label{lem:key}
Let $\alpha'=\gen{\St(\alpha)}{P}{z}\in\Genb{\alpha}$, and let
\[
Q=\{v\mid\prep(v)\in P\}\cup\{u\mid\pbody(u)\in P\}.
\]
Then $\beta=\gen{\alpha}{Q}{z}\in\Gen{\alpha}$, and there are a substitution $\theta$ and formulas
$\gamma_1,\dots,\gamma_m$ such that
\[
\theta\alpha'=(\gamma_1\to\gamma_1)\to\dots\to(\gamma_m\to\gamma_m)\to\beta .
\]
\end{lemma}
\begin{proof}
Let $\delta$ be the formula such that $P$ is a set of occurrences of
$\delta$. We first show that $\beta\in\Gen{\alpha}$. Clearly $Q\ne\emptyset$, and $z$ does not occur in $\alpha$,
since every variable of $\alpha$ occurs in $\St(\alpha)$. If $\delta$ is a
variable of $\alpha$, then by Lemma~\ref{lem:shape}(b), $Q$ is a set of
\kl{leaves} $v$ with $\subf{\alpha}{v}=\delta$, and some such \kl{leaf} is
not in $Q$, since $P$ does not contain all occurrences of $\delta$. If
$\delta=\ev{w}$, then $P=\{\prep(w)\}$ by Lemma~\ref{lem:shape}(b), since
$\alpha'\in\Genb{\alpha}$; hence $Q=\{w\}$, and $\subf{\alpha}{w}$ is not a
variable. If $\delta$ is not a variable, then
$Q$ is the set of $u$ with $\pbody(u)\in P$, so $\body{u}=\delta$ and
$\subf{\alpha}{u}=\decsub{\alpha}(\delta)$ for all $u\in Q$.

Since $\alpha'\in\Genb{\alpha}$, no occurrence $1^{i-1}$, $\pprem(u)$ or
$\pdef(u)$ belongs to $P$. In particular, for every \kl{non-leaf
occurrence} $v$ of $\alpha$,
\begin{equation}\label{eq:body}
\alpha'_{\pbody(v)}=
\begin{cases}
z, & \text{if $\pbody(v)\in P$},\\
\alpha'_{\prep(v0)}\to\alpha'_{\prep(v1)}, & \text{otherwise},
\end{cases}
\end{equation}
where $\alpha'_{\prep(vj)}$ is $z$ or $\rv{vj}$ for $j=0,1$. Let
$\theta(p)=p$ for every variable $p$ that is not an \kl{extension
variable}, and let
\[
\theta(\ev{v})=\theta(\alpha'_{\pbody(v)})
\]
for every \kl{non-leaf occurrence} $v$ of $\alpha$. By~\eqref{eq:body},
the only \kl{extension variables} that may occur in $\alpha'_{\pbody(v)}$
are $\ev{v0}$ and $\ev{v1}$, so this defines $\theta$ by induction on
$\hgt(\subf{\alpha}{v})$.

We claim that $\theta(\alpha'_{\prep(v)})=\subf{\beta}{v}$ for every
occurrence $v$ of $\alpha$ that is not \kl{strictly below} an element of
$Q$. Here $\prep(v)$ is an occurrence in $\alpha'$, since no proper prefix
of $\prep(v)$ belongs to $P$: the proper prefixes are of the form
$1^{i-1}$, together with $\pprem(w)$ and $\pbody(w)$ if $v=wj$, and in
the latter case $\pbody(w)\notin P$ because $w\notin Q$. We argue by
induction on $\hgt(\subf{\alpha}{v})$. If $\prep(v)\in P$, then $v\in Q$
and both sides are $z$. Otherwise $\alpha'_{\prep(v)}=\rv{v}$. If $v$ is a
\kl{leaf}, then $v\notin Q$ and both sides are $\subf{\alpha}{v}$. If $v$
is a \kl{non-leaf occurrence} and $\pbody(v)\in P$, then $v\in Q$ and
both sides are $z$. Otherwise $v\notin Q$, so neither $v0$ nor $v1$ is
\kl{strictly below} an element of $Q$, and by~\eqref{eq:body} and the
induction hypothesis,
\[
\theta(\ev{v})=\theta(\alpha'_{\prep(v0)})\to\theta(\alpha'_{\prep(v1)})
=\subf{\beta}{v0}\to\subf{\beta}{v1}=\subf{\beta}{v}.
\]

Since no occurrence $1^{i-1}$ belongs to $P$, we have
$\alpha'=\alpha'_{\pprem(u_1)}\to\dots\to\alpha'_{\pprem(u_m)}\to
\alpha'_{\prep(\rt)}$. Since $\pdef(u_i)\notin P$, the formula
$\alpha'_{\pprem(u_i)}$ is $\alpha'_{\pbody(u_i)}\to\ev{u_i}$ or
$\ev{u_i}\to\alpha'_{\pbody(u_i)}$, and $\theta$ maps it to
$\gamma_i\to\gamma_i$ with $\gamma_i=\theta(\ev{u_i})$. By the claim for
$v=\rt$, $\theta$ maps $\alpha'_{\prep(\rt)}$ to $\beta$.
\end{proof}

\begin{lemma}\label{lem:normal}
If $\Gen{\St(\alpha)}\cap L\neq\emptyset$ for a \kl{logic} $L$, then
$\Genb{\alpha}\cap L\neq\emptyset$.
\end{lemma}
\begin{proof}
Let $\alpha'=\gen{\St(\alpha)}{P}{z}\in L$, where $P$ is a set of
occurrences of a formula $\delta$. Since $\sigma\alpha'\in L$ for every
substitution $\sigma$ by (C1), it suffices to find $\sigma$ such that
$\sigma\alpha'\in\Genb{\alpha}$. We distinguish cases on $\delta$.

If $\delta$ is a variable of $\alpha$ or $\hgt(\delta)=1$, then every
element of $P$ is of the form $\prep(v)$ or $\pbody(u)$, respectively, by
Lemma~\ref{lem:shape}(b) and~(c), so $\alpha'\in\Genb{\alpha}$.

If $\delta=\ev{u}$, then $P=\{\prep(u)\}$ or $P=\{\pdef(u)\}$ by
Lemma~\ref{lem:shape}(b), since $P$ does not contain all occurrences of
$\delta$. In the first case $\alpha'\in\Genb{\alpha}$. In the second case,
the substitution that interchanges $\ev{u}$ and $z$ maps $\alpha'$ to
$\gen{\St(\alpha)}{\{\prep(u)\}}{z}$.

If $\hgt(\delta)=2$, then $\delta=\dprem{u}$ and $P=\{\pprem(u)\}$ for
some $u$ by Lemma~\ref{lem:shape}(c) and~(d). Let $d\in\{0,1\}$ be such
that $\pdef(u)=\pprem(u)d$. Substituting $\gen{\dprem{u}}{\{d\}}{z}$ for
$z$, we obtain $\gen{\St(\alpha)}{\{\pdef(u)\}}{z}$, which is treated in
the previous case.

If $\hgt(\delta)\ge3$, then $\delta=\tl{i}$ and $P=\{1^{i-1}\}$ for some
$i\le m$ by Lemma~\ref{lem:shape}(c) and~(d). Substituting
$\gen{\tl{i}}{\{1^{m-i+1}\}}{z}$ for $z$, we obtain
$\gen{\St(\alpha)}{\{\prep(\rt)\}}{z}\in\Genb{\alpha}$.
\end{proof}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
If $\alpha$ is a variable, then $\St(\alpha)=\alpha$ and there is
nothing to prove. Otherwise, suppose
that $\Gen{\St(\alpha)}\cap L\neq\emptyset$.
By Lemma~\ref{lem:normal}, there is $\alpha'\in\Genb{\alpha}\cap L$. Take $\beta$, $\theta$ and $\gamma_1,\dots,\gamma_m$ as in
Lemma~\ref{lem:key}. Then $\theta\alpha'\in L$ by (C1) and
$\gamma_i\to\gamma_i\in L$ by (C3), so $\beta\in L$ by (C2). This
contradicts the \kl{minimality} of $\alpha$ by
Proposition~\ref{cor:criterion}, since $\beta\in\Gen{\alpha}$. Hence
$\Gen{\St(\alpha)}\cap L=\emptyset$, and since $\St(\alpha)\in L$,
$\St(\alpha)$ is \kl{minimal} in $L$ by Proposition~\ref{cor:criterion}.
\end{proof}

\begin{example}\label{ex:decode}
Let $\alpha$ be Peirce's law as in Example~\ref{ex:peirce}, and replace
the occurrence of $\evfp{a\to b}$ in the \kl{antecedent}
$\evfn{(a\to b)\to a}\to\evfp{a\to b}\to a$, that is, at $\prep(00)$, by
$z$:
\[
\alpha'=((a\to b)\to\evfp{a\to b})\to(\evfn{(a\to b)\to a}\to z\to a)
\to((\evfn{(a\to b)\to a}\to a)\to\evfp{\alpha})\to\evfp{\alpha} .
\]
Then $Q=\{00\}$ and $\beta=(z\to a)\to a$. The substitution $\theta$ maps
$\evfp{a\to b}$ to $a\to b$, $\evfn{(a\to b)\to a}$ to $z\to a$ and $\evfp{\alpha}$ to
$(z\to a)\to a=\beta$, so that
\[
\theta\alpha'=((a\to b)\to(a\to b))\to((z\to a)\to(z\to a))\to
(\beta\to\beta)\to\beta .
\]
Since $\beta$ is not a classical tautology, neither is $\alpha'$.
\end{example}
